3// I/O port     Usage
4// 0x40         Channel 0 data port (read/write)
5// 0x41         Channel 1 data port (read/write)
6// 0x42         Channel 2 data port (read/write)
7// 0x43         Mode/Command register (write only, a read is ignored)
8const CHAN0_DATA: u8 = 0x40;
9const CHAN1_DATA: u8 = 0x41;
10const CHAN2_DATA: u8 = 0x42;
11const MODE_CMD: u8 = 0x43;
12
13// 	Bits         Usage
14// 	6 and 7      Select channel :
15// 	                0 0 = Channel 0
16// 	                0 1 = Channel 1
17// 	                1 0 = Channel 2
18// 	                1 1 = Read-back command (8254 only)
19// 	4 and 5      Access mode :
20// 	                0 0 = Latch count value command
21// 	                0 1 = Access mode: lobyte only
22// 	                1 0 = Access mode: hibyte only
23// 	                1 1 = Access mode: lobyte/hibyte
24// 	1 to 3       Operating mode :
25// 	                0 0 0 = Mode 0 (interrupt on terminal count)
26// 	                0 0 1 = Mode 1 (hardware re-triggerable one-shot)
27// 	                0 1 0 = Mode 2 (rate generator)
28// 	                0 1 1 = Mode 3 (square wave generator)
29// 	                1 0 0 = Mode 4 (software triggered strobe)
30// 	                1 0 1 = Mode 5 (hardware triggered strobe)
31// 	                1 1 0 = Mode 2 (rate generator, same as 010b)
32// 	                1 1 1 = Mode 3 (square wave generator, same as 011b)
33// 	0            BCD/Binary mode: 0 = 16-bit binary, 1 = four-digit BCD
34

54// System clock frequency
55pub static mut FREQUENCY: f64 = 1.0;
56pub static mut RELOAD_VALUE: u16 = 1;
57// Number of ms between each irq0
58
59//  Max frequency = 1193182, Min frequency = 18
60// frequency = 1193182 / reload_value;
61// frequency * reload_value = 1193182
62// reload_value = 1193182 / frequency
63//
64// frequency / 1000 = nb_cycle per ms
65// 1 / nb_cycle = ms between 2 cycle
66// ms correspond to the number of ms between two cycle we need
67pub fn set_irq0_in_ms(ms: f32) {
68	let mut frequency = 1000.0 / ms;
69
70	if frequency < 18.0 {
71		frequency = 18.0;
72	} else if frequency > 1193182.0 {
73		frequency = 1193182.0;
74	}
75
76	unsafe {
77		RELOAD_VALUE = (1193182.0 / frequency) as u16;
78		FREQUENCY = 1193182.0 / RELOAD_VALUE as f64;
79		// TODO fround
80		crate::time::SYSTEM_FRACTION = 1000.0 / FREQUENCY;
81		crate::kprintln!("System frequency set to: {}", FREQUENCY);
82		crate::kprintln!(
83			"System fraction set to: {}",
84			crate::time::SYSTEM_FRACTION
85		);
86		crate::kprintln!("Reload value set to: {:#x}", RELOAD_VALUE);
87		set_pit(CHANNEL_0, ACC_LOBHIB, MODE_2, RELOAD_VALUE);
88	}
89}

91pub fn set_pit(channel: u8, access: u8, mode: u8, data: u16) {
92	let port: u16 = match channel {
93		CHANNEL_0 => CHAN0_DATA.into(),
94		CHANNEL_1 => CHAN1_DATA.into(),
95		CHANNEL_2 => CHAN2_DATA.into(),
96		_ => panic!("PIT Channel {channel} doesn't exist")
97	};
98	// Set cmd mod selected
99	outb(MODE_CMD as u16, channel | access | mode);
100
101	// Sending data to commands
102	outb(port, (data & 0xff) as u8);
103	outb(port, ((data & 0xff00) >> 8) as u8);
104}

106pub fn play_sound(frequency: f32) {
107	let freq = (1193182.0 / frequency) as u16;
108	outb(0x43, 0xb6);
109	outb(0x42, (freq & 0xff) as u8);
110	outb(0x42, ((freq & 0xff00) >> 8) as u8);
111}

113pub fn speaker_on() {
114	let tmp: u8 = inb(0x61);
115	if tmp != (tmp | 3) {
116		outb(0x61, tmp | 3);
117	}
118}
