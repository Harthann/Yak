12// 	PIC2 Interrupt		BIT	 ____________		PIC1 Interrupt		BIT	 ____________
13// Real Time Clock -->	[0]	|			 |   Timer ------------->	[0]	|			 |
14// ACPI ------------->	[1]	|            |   Keyboard----------->	[1]	|            |      _____
15// Available -------->	[2]	| Secondary  |---------------------->	[2]	| Primary    |     |     |
16// Available -------->	[3]	| Interrupt  |   Serial Port 2 ----->	[3]	| Interrupt  |---> | CPU |
17// Mouse ------------>	[4]	| Controller |   Serial Port 1 ----->	[4]	| Controller |     |_____|
18// Co-Processor ----->	[5]	|            |   Parallel Port 2/3 ->	[5]	|            |
19// Primary ATA ------>	[6]	|            |   Floppy disk ------->	[6]	|            |
20// Secondary ATA ---->	[7]	|____________|   Parallel Port 1---->	[7]	|____________|
21
22#[repr(u8)]
23pub enum Pic1 {
24	Timer          = 0x01, // 0b0000 0001
25	Keyboard       = 0x02, // 0b0000 0010
26	Slave          = 0x04, // 0b0000 0100
27	Serial1        = 0x08, // 0b0000 1000
28	Serial2        = 0x10, // 0b0001 0000
29	ParallelPort23 = 0x20, // 0b0010 0000
30	FloppyDisk     = 0x40, // 0b0100 0000
31	ParallelPort1  = 0x80  // 0b1000 0000
32}

34#[repr(u8)]
35pub enum Pic2 {
36	RealTimeClock = 0x01, // 0b0000 0001
37	ACPI          = 0x02, // 0b0000 0010
38	Available0    = 0x04, // 0b0000 0100
39	Available1    = 0x08, // 0b0000 1000
40	Mouse         = 0x10, // 0b0001 0000
41	CoProcessor   = 0x20, // 0b0010 0000
42	PrimaryATA    = 0x40, // 0b0100 0000
43	SecondaryATA  = 0x80  // 0b1000 0000
44}

75#[inline]
76#[no_mangle]
77pub fn end_of_interrupts(irq: usize) {
78	if irq >= 8 {
79		// Slave interrupt request
80		outb(PIC2_CMD, PIC_EOI);
81	}
82	outb(PIC1_CMD, PIC_EOI);
83}

85const ICW1_ICW4: u8 = 0x01; // ICW4 (not needed)
86const ICW1_SINGLE: u8 = 0x02; // Single (cascade) mode
87const ICW1_INTERVAL4: u8 = 0x04; // Call address interval 4 (8)
88const ICW1_LEVEL: u8 = 0x08; // Level triggered (edge) mode
89const ICW1_INIT: u8 = 0x10; // Initiliazation - required!
90
91const ICW4_8086: u8 = 0x01; // 8086/88 (MCS-80/85) mode
92const ICW4_AUTO: u8 = 0x02; // Auto (normal EOI)
93const ICW4_BUF_SLAVE: u8 = 0x08; // Buffered mode/slave
94const ICW4_BUF_MASTER: u8 = 0x0c; // Buffered mode/master
95const ICW4_SFNM: u8 = 0x10; // Special fully nested (not)

97pub fn pic_remap(offset1: u8, offset2: u8) {
98	// save masks
99	let a1: u8 = inb(PIC1_DATA);
100	let a2: u8 = inb(PIC2_DATA);
101
102	// Init Control Word 1, tell to pic we start init process
103	outb(PIC1_CMD, ICW1_INIT | ICW1_ICW4);
104	io_wait();
105	outb(PIC2_CMD, ICW1_INIT | ICW1_ICW4);
106	io_wait();
107
108	// Init Control Word 2, tell to pic offset of interrupt
109	outb(PIC1_DATA, offset1);
110	io_wait();
111	outb(PIC2_DATA, offset2);
112	io_wait();
113
114	// Init Control Word 3, tell to pic who's master/slave
115	outb(PIC1_DATA, 4);
116	io_wait();
117	outb(PIC2_DATA, 2);
118	io_wait();
119
120	// Init Control Word 4, tell to pic we use mode 8086
121	outb(PIC1_DATA, ICW4_8086);
122	io_wait();
123	outb(PIC2_DATA, ICW4_8086);
124	io_wait();
125
126	// restore masks
127	outb(PIC1_DATA, a1);
128	outb(PIC2_DATA, a2);
129}

131pub fn irq_set_mask(mut irq: usize) {
132	let port: u16;
133	let value: u8;
134
135	if irq < 8 {
136		port = PIC1_DATA;
137	} else {
138		port = PIC2_DATA;
139		irq -= 8;
140	}
141	value = inb(port) | (1 << irq);
142	outb(port, value);
143}
144
145pub fn irq_clear_mask(mut irq: usize) {
146	let port: u16;
147	let value: u8;
148
149	if irq < 8 {
150		port = PIC1_DATA;
151	} else {
152		port = PIC2_DATA;
153		irq -= 8;
154	}
155	value = inb(port) & !(1 << irq);
156	outb(port, value);
157}

159pub fn pic_init_masks() {
160	// To activate specific irqs
161	let pic1mask: u8 =
162		Pic1::Timer as u8 | Pic1::Keyboard as u8 | Pic1::Slave as u8;
163
164	let pic2mask: u8 = Pic2::RealTimeClock as u8;
165
166	// We need the bit to be 0 to activate
167	outb(PIC1_DATA, 0xff ^ pic1mask);
168	outb(PIC2_DATA, 0xff ^ pic2mask);
169}
170
171pub fn setup_pic8259() {
172	pic_init_masks();
173	pic_remap(PIC1_IRQ_OFFSET, PIC2_IRQ_OFFSET);
