1// mount_no indicate the number of mount since last fsck
2// mount_no_max indicate the number of mount between each fsck
3/// Superblock always take 1024 bytes with/without Extended block
4#[derive(Default, Debug, PartialEq, Eq)]
5pub struct BaseSuperblock {
6	/// Total number of inodes in file system
7	inode_count:          u32,
8	/// Total number of blocks in file system
9	blocks_count:         u32,
10	/// Number of blocks reserved for superuser (see offset 80)
11	rblocks_num:          u32,
12	/// Total number of unallocated blocks
13	pub blocks_unalloc:   u32,
14	/// Total number of unallocated inodes
15	pub inode_unalloc:    u32,
16	/// Block number of the block containing the superblock (also the starting block number, NOT always zero.)
17	pub superblock_block: u32,
18	/// log2 (block size) - 10. (In other words, the number to shift 1,024 to the left by to obtain the block size)
19	block_size:           u32,
20	/// log2 (fragment size) - 10. (In other words, the number to shift 1,024 to the left by to obtain the fragment size)
21	frag_size:            u32,
22	/// Number of blocks in each block group
23	bgroup_bno:           u32,
24	/// Number of fragments in each block group
25	bgroup_fno:           u32,
26	/// Number of inodes in each block group
27	bgroup_ino:           u32,
28	/// Last mount time (in POSIX time)
29	last_mt:              u32,
30	/// Last written time (in POSIX time)
31	last_wt:              u32,
32	/// Number of times the volume has been mounted since its last consistency check (fsck)
33	mount_no:             u16,
34	/// Number of mounts allowed before a consistency check (fsck) must be done
35	mount_no_max:         u16,
36	/// Ext2 signature (0xef53), used to help confirm the presence of Ext2 on a volume
37	ext2_sig:             u16,
38	/// File system state (see below)
39	fs_state:             u16,
40	/// What to do when an error is detected (see below)
41	err_handle:           u16,
42	/// Minor portion of version (combine with Major portion below to construct full version field)
43	minor:                u16,
44	/// POSIX time of last consistency check (fsck)
45	last_fsck:            u32,
46	/// Interval (in POSIX time) between forced consistency checks (fsck)
47	fsck_interval:        u32,
48	/// Operating system ID from which the filesystem on this volume was created (see below)
49	osid:                 u32,
50	/// Major portion of version (combine with Minor portion above to construct full version field)
51	major:                u32,
52	/// User ID that can use reserved blocks
53	pub uid:              u16,
54	/// Group ID that can use reserved blocks
55	pub gid:              u16,
56	extension:            Option<ExtendedSuperblock>
57}

59impl BaseSuperblock {
60	pub fn sig(&self) -> u16 {
61		self.ext2_sig
62	}
63
64	pub fn version(&self) -> (u32, u16) {
65		(self.major, self.minor)
66	}
67
68	pub fn bsize(&self) -> usize {
69		1024 << self.block_size
70	}
71	pub fn inode_size(&self) -> u16 {
72		if self.major < 1 {
73			return 128;
74		}
75		match &self.extension {
76			Some(ext) => ext.inode_size,
77			_ => panic!("Superblock extension not set with major >= 1")
78		}
79	}
80
81	pub fn block_per_grp(&self) -> u32 {
82		self.bgroup_bno
83	}
84
85	pub fn inode_per_grp(&self) -> u32 {
86		self.bgroup_ino
87	}
88
89	pub fn inode_count(&self) -> u32 {
90		self.inode_count
91	}
92
93	pub fn block_count(&self) -> u32 {
94		self.blocks_count
95	}
96
97	pub fn block_grp_count(&self) -> u32 {
98		(self.blocks_count / self.bgroup_bno)
99			+ (self.blocks_count % self.bgroup_bno != 0) as u32
100	}
101
102	pub fn set_extension(&mut self, extension: ExtendedSuperblock) {
103		self.extension = Some(extension);
104	}
105}

128			),
129			superblock_block: u32::from_le_bytes(

174			fsck_interval:    u32::from_le_bytes(

176			),
177			osid:             u32::from_le_bytes(
178				buffer[72..76].try_into().unwrap()
179			),
180			major:            u32::from_le_bytes(
181				buffer[76..80].try_into().unwrap()
182			),
183			uid:              u16::from_le_bytes(
184				buffer[80..82].try_into().unwrap()
185			),
186			gid:              u16::from_le_bytes(
187				buffer[82..84].try_into().unwrap()

189			extension:        None

191	}

228use core::fmt;
229impl fmt::Display for BaseSuperblock {
230	fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
231		let lwt = crate::time::ctime(self.last_wt);
232		let lct = crate::time::ctime(self.last_fsck);
233		write!(
234			f,
235			"Superblock: {{
236Sig: {:#x}
237Version: {}.{}
238Block Size: {:#x}
239Last write time: {lwt},
240Last check time: {lct},
241}}",
242			self.ext2_sig,
243			self.major,
244			self.minor,
245			1024 << self.block_size,
246		)
247	}
248}

263/// Present if Major >= 1
264/// Bytes from 236 to 1023 aren't used
265#[derive(Debug, PartialEq, Eq)]
266pub struct ExtendedSuperblock {
267	/// First non-reserved inode in file system. (In versions < 1.0, this is fixed as 11)
268	first_inode:           u32,
269	/// Size of each inode structure in bytes. (In versions < 1.0, this is fixed as 128)
270	inode_size:            u16,
271	/// Block group that this superblock is part of (if backup copy)
272	bgroup_superblock:     u16,
273	/// Optional features present (features that are not required to read or write, but usually result in a performance increase. see below)
274	opt_features:          u32,
275	/// Required features present (features that are required to be supported to read or write. see below)
276	req_features:          u32,
277	/// Features that if not supported, the volume must be mounted read-only see below)
278	ro_features:           u32,
279	/// File system ID (what is output by blkid)
280	fsid:                  [u8; 16],
281	/// Volume name (C-style string: characters terminated by a 0 byte)
282	vol_name:              [u8; 16],
283	/// Path volume was last mounted to (C-style string: characters terminated by a 0 byte)
284	last_path:             [u8; 64],
285	/// Compression algorithms used (see Required features above)
286	compr:                 u32,
287	/// Number of blocks to preallocate for files
288	prealloc_blocks_files: u8,
289	/// Number of blocks to preallocate for directories
290	prealloc_block_dir:    u8,
291	/// (Unused)
292	unused:                u16,
293	/// Journal ID (same style as the File system ID above)
294	journ_id:              [u8; 16],
295	/// Journal inode
296	journ_inode:           u32,
297	/// Journal device
298	journ_dev:             u32,
299	/// Head of orphan inode list
300	orphan_inode_lst:      u32
301}

303impl From<&[u8]> for ExtendedSuperblock {
304	fn from(buffer: &[u8]) -> Self {
305		if buffer.len() != 152 {
306			panic!("Wrong size for Extended Super block parsing");
307		}
308		let mut exblock = Self {
309			first_inode:           u32::from_le_bytes(
310				buffer[0..4].try_into().unwrap()
311			),
312			inode_size:            u16::from_le_bytes(
313				buffer[4..6].try_into().unwrap()
314			),
315			bgroup_superblock:     u16::from_le_bytes(
316				buffer[6..8].try_into().unwrap()
317			),
318			opt_features:          u32::from_le_bytes(
319				buffer[8..12].try_into().unwrap()
320			),
321			req_features:          u32::from_le_bytes(
322				buffer[12..16].try_into().unwrap()
323			),
324			ro_features:           u32::from_le_bytes(
325				buffer[16..20].try_into().unwrap()
326			),
327			fsid:                  [0; 16],
328			vol_name:              [0; 16],
329			last_path:             [0; 64],
330			compr:                 u32::from_le_bytes(
331				buffer[116..120].try_into().unwrap()
332			),
333			prealloc_blocks_files: buffer[120],
334			prealloc_block_dir:    buffer[121],
335			unused:                0x0,
336			journ_id:              [0; 16],
337			journ_inode:           u32::from_le_bytes(
338				buffer[140..144].try_into().unwrap()
339			),
340			journ_dev:             u32::from_le_bytes(
341				buffer[144..148].try_into().unwrap()
342			),
343			orphan_inode_lst:      u32::from_le_bytes(
344				buffer[148..152].try_into().unwrap()
345			)
346		};
347		exblock.fsid[0..16].copy_from_slice(&buffer[20..36]);
348		exblock.vol_name[0..16].copy_from_slice(&buffer[36..52]);
349		exblock.last_path[0..64].copy_from_slice(&buffer[52..116]);
350		exblock.journ_id[0..16].copy_from_slice(&buffer[124..140]);
351
352		exblock
353	}
354}
