5/// Like blocks, each inode has a numerical address.
6/// It is extremely important to note that unlike block addresses, inode addresses start at 1.
7/// With Ext2 versions prior to Major version 1, inodes 1 to 10 are reserved and
8/// should be in an allocated state.
9/// Starting with version 1, the first non-reserved inode is indicated via a field in the Superblock.
10/// Of the reserved inodes, number 2 subjectively has the most significance as it is used for the root directory.
11/// Inodes have a fixed size of either 128 for version 0 Ext2 file systems, or as dictated by the field in the Superblock for version 1 file systems.
12/// All inodes reside in inode tables that belong to block groups.
13/// Therefore, looking up an inode is simply a matter of determining which
14/// block group it belongs to and indexing that block group's inode table.
15#[derive(Debug, Default, Copy, Clone)]
16pub struct Inode {
17	/// Type and Permissions (see below)
18	pub tperm:      u16,
19	/// User ID
20	pub uid:        u16,
21	/// Lower 32 bits of size in bytes
22	pub size_lh:    u32,
23	/// Last Access Time (in POSIX time)
24	pub lat:        u32,
25	/// Creation Time (in POSIX time)
26	pub creatt:     u32,
27	/// Last Modification time (in POSIX time)
28	pub lmt:        u32,
29	/// Deletion time (in POSIX time)
30	pub delt:       u32,
31	/// Group ID
32	pub gid:        u16,
33	/// Count of hard links (directory entries) to this inode. When this reaches 0, the data blocks are marked as unallocated.
34	pub count_hl:   u16,
35	/// Count of disk sectors (not Ext2 blocks) in use by this inode, not counting the actual inode structure nor directory entries linking to the inode.
36	pub count_ds:   u32,
37	/// Flags (see below)
38	pub flags:      u32,
39	/// Operating System Specific value #1
40	os_specific1:   u32,
41	/// Direct Block Pointers
42	pub dbp:        [u32; 12],
43	/// Singly Indirect Block Pointer (Points to a block that is a list of block pointers to data)
44	pub sibp:       u32,
45	/// Doubly Indirect Block Pointer (Points to a block that is a list of block pointers to Singly Indirect Blocks)
46	pub dibp:       u32,
47	/// Triply Indirect Block Pointer (Points to a block that is a list of block pointers to Doubly Indirect Blocks)
48	pub tibp:       u32,
49	/// Generation number (Primarily used for NFS)
50	pub gen_no:     u32,
51	/// In Ext2 version 0, this field is reserved. In version >= 1, Extended attribute block (File ACL).
52	pub facl:       u32,
53	/// In Ext2 version 0, this field is reserved. In version >= 1, Upper 32 bits of file size (if feature bit set) if it's a file, Directory ACL if it's a directory
54	pub size_uh:    u32,
55	/// Block address of fragment
56	pub block_addr: u32,
57	/// Operating System Specific Value #2
58	os_specific2:   [u8; 12]
59}

90	pub fn size(&self) -> u64 {
91		self.size_lh as u64 | ((self.size_uh as u64) << 32)
92	}

107	pub fn is_valid_block(block_no: u32) -> bool {
108		block_no != 0
109	}

111	fn get_blocks_no_from_u32_slice(slice: &[u32]) -> Vec<u32> {
112		let mut blocks_no = Vec::new();
113		for i in 0..slice.len() {
114			if Inode::is_valid_block(slice[i]) {
115				blocks_no.push(slice[i]);
116			}
117		}
118		blocks_no
119	}
120
121	fn get_blocks_no_from_u8_slice(slice: &[u8]) -> Vec<u32> {
122		let mut blocks_no: Vec<u32> = Vec::new();
123		for i in (0..slice.len()).step_by(size_of::<u32>()) {
124			let block_no = u32::from_le_bytes(
125				slice[i..i + size_of::<u32>()].try_into().unwrap()
126			);
127			if Inode::is_valid_block(block_no) {
128				blocks_no.push(block_no);
129			}
130		}
131		blocks_no
132	}

134	pub fn get_blocks_no(&self) -> Vec<u32> {
135		Inode::get_blocks_no_from_u32_slice(&self.dbp)
136	}

138	pub fn _get_sibp_blocks_no(sibp: u32, ext2: &super::Ext2) -> Vec<u32> {
139		let singly_block = ext2.read_block(sibp);
140		Inode::get_blocks_no_from_u8_slice(&singly_block)

143	pub fn get_sibp_blocks_no(&self, ext2: &super::Ext2) -> Vec<u32> {
144		Inode::_get_sibp_blocks_no(self.sibp, ext2)
145	}
146
147	pub fn _get_dibp_blocks_no(dibp: u32, ext2: &super::Ext2) -> Vec<u32> {
148		let doubly_block = ext2.read_block(dibp);
149		let mut blocks_no: Vec<u32> = Vec::new();
150		let singly_blocks = Inode::get_blocks_no_from_u8_slice(&doubly_block);
151		for singly_block in singly_blocks {
152			blocks_no.extend_from_slice(&Inode::_get_sibp_blocks_no(
153				singly_block,
154				ext2
155			));
156		}
157		blocks_no
158	}

172			));

182impl From<&[u8]> for Inode {
183	fn from(buffer: &[u8]) -> Self {
184		if buffer.len() < 128 {
185			panic!("Wrong size while converting slice to Superblock");
186		}
187		// Safe beceause len is forced to be 83
188		let mut inode = Self {
189			tperm:        u16::from_le_bytes(buffer[0..2].try_into().unwrap()),
190			uid:          u16::from_le_bytes(buffer[2..4].try_into().unwrap()),
191			size_lh:      u32::from_le_bytes(buffer[4..8].try_into().unwrap()),
192			lat:          u32::from_le_bytes(buffer[8..12].try_into().unwrap()),
193			creatt:       u32::from_le_bytes(
194				buffer[12..16].try_into().unwrap()
195			),
196			lmt:          u32::from_le_bytes(
197				buffer[16..20].try_into().unwrap()
198			),
199			delt:         u32::from_le_bytes(
200				buffer[20..24].try_into().unwrap()
201			),
202			gid:          u16::from_le_bytes(
203				buffer[24..26].try_into().unwrap()
204			),
205			count_hl:     u16::from_le_bytes(
206				buffer[26..28].try_into().unwrap()
207			),
208			count_ds:     u32::from_le_bytes(
209				buffer[28..32].try_into().unwrap()
210			),
211			flags:        u32::from_le_bytes(
212				buffer[32..36].try_into().unwrap()
213			),
214			os_specific1: u32::from_le_bytes(
215				buffer[36..40].try_into().unwrap()
216			),
217			dbp:          [
218				u32::from_le_bytes(buffer[40..44].try_into().unwrap()),
219				u32::from_le_bytes(buffer[44..48].try_into().unwrap()),
220				u32::from_le_bytes(buffer[48..52].try_into().unwrap()),
221				u32::from_le_bytes(buffer[52..56].try_into().unwrap()),
222				u32::from_le_bytes(buffer[56..60].try_into().unwrap()),
223				u32::from_le_bytes(buffer[60..64].try_into().unwrap()),
224				u32::from_le_bytes(buffer[64..68].try_into().unwrap()),
225				u32::from_le_bytes(buffer[68..72].try_into().unwrap()),
226				u32::from_le_bytes(buffer[72..76].try_into().unwrap()),
227				u32::from_le_bytes(buffer[76..80].try_into().unwrap()),
228				u32::from_le_bytes(buffer[80..84].try_into().unwrap()),
229				u32::from_le_bytes(buffer[84..88].try_into().unwrap())
230			],
231			sibp:         u32::from_le_bytes(
232				buffer[88..92].try_into().unwrap()
233			),
234			dibp:         u32::from_le_bytes(
235				buffer[92..96].try_into().unwrap()
236			),
237			tibp:         u32::from_le_bytes(
238				buffer[96..100].try_into().unwrap()
239			),
240			gen_no:       u32::from_le_bytes(
241				buffer[100..104].try_into().unwrap()
242			),
243			facl:         u32::from_le_bytes(
244				buffer[104..108].try_into().unwrap()
245			),
246			size_uh:      u32::from_le_bytes(
247				buffer[108..112].try_into().unwrap()
248			),
249			block_addr:   u32::from_le_bytes(
250				buffer[112..116].try_into().unwrap()
251			),
252			os_specific2: [0; 12]
253		};
254		inode.os_specific2[0..12].copy_from_slice(&buffer[116..128]);
255		inode
256	}

298use core::fmt;
299impl fmt::Display for Inode {
300	fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
301		let size: u64 = self.size();
302
303		let mut perms: crate::string::String = crate::string::String::new();
304		if self.tperm & IPERM_OREAD != 0 {
305			perms.push('r');
306		} else {
307			perms.push('-');
308		}
309		if self.tperm & IPERM_OWRIT != 0 {
310			perms.push('w');
311		} else {
312			perms.push('-');
313		}
314		if self.tperm & IPERM_OEXEC != 0 {
315			perms.push('x');
316		} else {
317			perms.push('-');
318		}
319
320		if self.tperm & IPERM_GREAD != 0 {
321			perms.push('r');
322		} else {
323			perms.push('-');
324		}
325		if self.tperm & IPERM_GWRIT != 0 {
326			perms.push('w');
327		} else {
328			perms.push('-');
329		}
330		if self.tperm & IPERM_GEXEC != 0 {
331			perms.push('x');
332		} else {
333			perms.push('-');
334		}
335
336		if self.tperm & IPERM_UREAD != 0 {
337			perms.push('r');
338		} else {
339			perms.push('-');
340		}
341		if self.tperm & IPERM_UWRIT != 0 {
342			perms.push('w');
343		} else {
344			perms.push('-');
345		}
346		if self.tperm & IPERM_UEXEC != 0 {
347			perms.push('x');
348		} else {
349			perms.push('-');
350		}
351
352		let date = crate::time::ctime(self.lmt);
353		write!(
354			f,
355			"{perms} {hardlinks} {uid} {gid} {size:>4} {date}",
356			hardlinks = self.count_hl,
357			uid = self.uid,
358			gid = self.gid
359		)
360	}

419use crate::alloc::string::ToString;
420impl From<&[u8]> for Dentry {
421	fn from(buffer: &[u8]) -> Self {
422		let mut dentry = Self {
423			inode:       u32::from_le_bytes(buffer[0..4].try_into().unwrap()),
424			dentry_size: u16::from_le_bytes(buffer[4..6].try_into().unwrap()),
425			name_length: u8::from_le_bytes(buffer[6..7].try_into().unwrap()),
426			r#type:      u8::from_le_bytes(buffer[7..8].try_into().unwrap()),
427			name:        crate::string::String::new()
428		};
429		if dentry.name_length != 0 {
430			dentry.name = core::str::from_utf8(
431				&buffer[8..8 + dentry.name_length as usize]
432			)
433			.expect("Error")
434			.to_string();
435		}
436		dentry
437	}
438}
