14enum BeatType {
15	WHOLE     = 1,
16	HALF      = 2,
17	QUARTER   = 4,
18	EIGTH     = 8,
19	SIXTEENTH = 16,
20	THIRTY2ND = 32
21}

23struct Partition {
24	bpm:                     usize,
25	whole_note_duration:     usize,
26	half_note_duration:      usize,
27	quarter_note_duration:   usize,
28	eigth_note_duration:     usize,
29	sixteenth_note_duration: usize,
30	thirty2nd_note_duration: usize,
31	notes:                   Vec<Note>
32}
33
34impl Partition {
35	pub fn new(bpm: usize, beat_type: BeatType) -> Self {
36		let whole_note = (60000 / bpm) * beat_type as usize;
37		Partition {
38			bpm:                     bpm,
39			whole_note_duration:     whole_note,
40			half_note_duration:      whole_note / 2,
41			quarter_note_duration:   whole_note / 4,
42			eigth_note_duration:     whole_note / 8,
43			sixteenth_note_duration: whole_note / 16,
44			thirty2nd_note_duration: whole_note / 32,
45			notes:                   Vec::new()
46		}
47	}
48
49	pub fn add_note(
50		&mut self,
51		frequency: f32,
52		note_tempo: NoteTempo,
53		note_type: NoteType
54	) -> &mut Self {
55		let duration: usize = match note_tempo {
56			NoteTempo::WHOLE => self.whole_note_duration,
57			NoteTempo::HALF => self.half_note_duration,
58			NoteTempo::QUARTER => self.quarter_note_duration,
59			NoteTempo::EIGTH => self.eigth_note_duration,
60			NoteTempo::SIXTEENTH => self.sixteenth_note_duration,
61			NoteTempo::THIRTY2ND => self.thirty2nd_note_duration
62		};
63		match note_type {
64			NoteType::BASE => self.notes.push(Note::new(frequency, duration)),
65			NoteType::DOTTED => {
66				self.notes.push(Note::new(frequency, duration * 3 / 2))
67			},
68			NoteType::TRIPLET => {
69				self.notes.push(Note::new(frequency, duration / 3));
70				self.notes.push(Note::new(frequency, duration / 3));
71				self.notes.push(Note::new(frequency, duration / 3));
72			}
73		}
74		self
75	}
76
77	pub fn add_triplet(
78		&mut self,
79		frequencies: (f32, f32, f32),
80		note_tempo: NoteTempo
81	) {
82		let duration: usize = match note_tempo {
83			NoteTempo::WHOLE => self.whole_note_duration,
84			NoteTempo::HALF => self.half_note_duration,
85			NoteTempo::QUARTER => self.quarter_note_duration,
86			NoteTempo::EIGTH => self.eigth_note_duration,
87			NoteTempo::SIXTEENTH => self.sixteenth_note_duration,
88			NoteTempo::THIRTY2ND => self.thirty2nd_note_duration
89		};
90		self.notes.push(Note::new(frequencies.0, duration / 3));
91		self.notes.push(Note::new(frequencies.1, duration / 3));
92		self.notes.push(Note::new(frequencies.2, duration / 3));
93	}
94
95	pub fn add_irregular_note(
96		&mut self,
97		frequency: f32,
98		note_tempo: NoteTempo,
99		div: usize
100	) {
101		let duration: usize = match note_tempo {
102			NoteTempo::WHOLE => self.whole_note_duration,
103			NoteTempo::HALF => self.half_note_duration,
104			NoteTempo::QUARTER => self.quarter_note_duration,
105			NoteTempo::EIGTH => self.eigth_note_duration,
106			NoteTempo::SIXTEENTH => self.sixteenth_note_duration,
107			NoteTempo::THIRTY2ND => self.thirty2nd_note_duration
108		};
109		self.notes.push(Note::new(frequency, duration / div));
110	}
111
112	pub fn play(&self) {
113		speaker_on();
114		for i in self.notes.iter() {
115			i.play();
116		}
117		speaker_off();
118	}
119}
