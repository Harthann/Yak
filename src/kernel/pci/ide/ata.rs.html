103impl ATA {
104	pub fn access(
105		direction: u8,
106		device: &IDEDevice,
107		lba: u32,
108		numsects: u8,
109		mut edi: u32
110	) -> Result<(), u8> {
111		let binding = match &device.channel {
112			Some(x) => x,
113			None => return Err(0x1)
114		};
115		let bind = binding.lock();
116		let mut channel = bind.borrow_mut();
117		let lba_mode: u8; // 0: CHS, 1: LBA28, 2: LBA48
118		let dma: u8; // 0: No DMA, 1: DMA
119		let mut lba_io: [u8; 6] = [0; 6];
120		// Read the Drive [Master/Slave]
121		let slavebit: u32 = device.drive as u32;
122		// Bus Base, like 0x1f0 which is also data port
123		let bus: u32 = channel.base as u32;
124		// Almost every ATA drive has sector-size of 512-byte
125		let words: u32 = SECTOR_SIZE / 2;
126		let head: u8;
127
128		// Disable IRQ
129		*IDE_IRQ_INVOKED.lock() = 0x0;
130		channel.n_ien = 0x02;
131		channel.write(ATAReg::CONTROL, 0x02);
132
133		// (I) Select one from LBA28, LBA48 or CHS
134		// Sure Drive should support LBA in this case or you
135		// are giving a wrong LBA
136		if lba >= 0x10000000 {
137			// LBA48
138			lba_mode = 2;
139			lba_io[0] = ((lba & 0x000000FF) >> 0) as u8;
140			lba_io[1] = ((lba & 0x0000FF00) >> 8) as u8;
141			lba_io[2] = ((lba & 0x00FF0000) >> 16) as u8;
142			lba_io[3] = ((lba & 0xFF000000) >> 24) as u8;
143			lba_io[4] = 0; // LBA28 is integer, so 32-bits are enough to access 2TB
144			lba_io[5] = 0; // LBA28 is integer, so 32-bits are enough to access 2TB
145			head = 0; // Lower 4-bits of HDDEVSEL are not used here
146		} else if device.capabilities & 0x200 != 0 {
147			// LBA48
148			lba_mode = 1;
149			lba_io[0] = ((lba & 0x00000FF) >> 0) as u8;
150			lba_io[1] = ((lba & 0x000FF00) >> 8) as u8;
151			lba_io[2] = ((lba & 0x0FF0000) >> 16) as u8;
152			lba_io[3] = 0; // These Registers are not used here
153			lba_io[4] = 0; // These Registers are not used here
154			lba_io[5] = 0; // These Registers are not used here
155			head = ((lba & 0xF000000) >> 24) as u8;
156		} else {
157			// CHS:
158			lba_mode = 0;
159			let sect: u8 = ((lba % 63) + 1) as u8;
160			let cyl: u16 = ((lba + 1 - sect as u32) / (16 * 63)) as u16;
161			lba_io[0] = sect;
162			lba_io[1] = ((cyl >> 0) & 0xFF) as u8;
163			lba_io[2] = ((cyl >> 8) & 0xFF) as u8;
164			lba_io[3] = 0;
165			lba_io[4] = 0;
166			lba_io[5] = 0;
167			// Head number is written to HDDEVSEL lower 4-bits
168			head = ((lba + 1 - sect as u32) % (16 * 63) / (63)) as u8;
169		}
170
171		// (II) See if drive supports DMA or not
172		dma = 0; // We don't support DMA
173
174		// (III) Wait if the drive is busy
175		while (channel.read(ATAReg::STATUS) & ATAStatus::BSY) != 0 {}
176
177		// (IV) Select Drive from the controller
178		if lba_mode == 0 {
179			// Drive & CHS
180			channel
181				.write(ATAReg::HDDEVSEL, 0xa0 | ((slavebit as u8) << 4) | head);
182		} else {
183			// Drive & LBA
184			channel
185				.write(ATAReg::HDDEVSEL, 0xe0 | ((slavebit as u8) << 4) | head);
186		}
187
188		// (V) Write Parameters
189		if lba_mode == 2 {
190			channel.write(ATAReg::SECCOUNT1, 0);
191			channel.write(ATAReg::LBA3, lba_io[3]);
192			channel.write(ATAReg::LBA4, lba_io[4]);
193			channel.write(ATAReg::LBA5, lba_io[5]);
194		}
195		channel.write(ATAReg::SECCOUNT0, numsects);
196		channel.write(ATAReg::LBA0, lba_io[0]);
197		channel.write(ATAReg::LBA1, lba_io[1]);
198		channel.write(ATAReg::LBA2, lba_io[2]);
199
200		// (VI) Select the command and send it
201		// Routine that is followed:
202		// If ( DMA & LBA48)   DO_DMA_EXT
203		// If ( DMA & LBA28)   DO_DMA_LBA
204		// If ( DMA & LBA28)   DO_DMA_CHS
205		// If (!DMA & LBA48)   DO_PIO_EXT
206		// If (!DMA & LBA28)   DO_PIO_LBA
207		// If (!DMA & !LBA#)   DO_PIO_CHS
208
209		let cmd = match (lba_mode, dma, direction) {
210			(0, 0, 0) => ATACommand::ReadPio,
211			(1, 0, 0) => ATACommand::ReadPio,
212			(2, 0, 0) => ATACommand::ReadPioExt,
213			(0, 1, 0) => ATACommand::ReadDma,
214			(1, 1, 0) => ATACommand::ReadDma,
215			(2, 1, 0) => ATACommand::ReadDmaExt,
216			(0, 0, 1) => ATACommand::WritePio,
217			(1, 0, 1) => ATACommand::WritePio,
218			(2, 0, 1) => ATACommand::WritePioExt,
219			(0, 1, 1) => ATACommand::WriteDma,
220			(1, 1, 1) => ATACommand::WriteDma,
221			(2, 1, 1) => ATACommand::WriteDmaExt,
222			_ => todo!()
223		};
224		// Send the command
225		channel.write(ATAReg::COMMAND, cmd as u8);
226
227		if dma != 0 {
228			if direction == 0 {
229				// DMA Read
230			} else {
231				// DMA Write
232			}
233		} else {
234			if direction == 0 {
235				// PIO Read
236				for _ in 0..numsects {
237					// Polling, set error and exit if there is
238					channel.polling(1)?;
239					io::insw(bus as u16, edi as *mut _, words);
240					edi += words * 2;
241				}
242			} else {
243				// PIO Write
244				for _ in 0..numsects {
245					// Polling
246					channel.polling(0)?;
247					io::outsw(bus as u16, edi as *mut _, words);
248					edi += words * 2;
249				}
250				channel.write(
251					ATAReg::COMMAND,
252					[
253						ATACommand::CacheFlush,
254						ATACommand::CacheFlush,
255						ATACommand::CacheFlushExt
256					][lba_mode as usize] as u8
257				);
258				// Polling
259				channel.polling(0)?;
260			}
261		}
262		Ok(())
263	}
