35use crate::cli::DISKNO;
36use crate::disk;
37use crate::fs::ext2;
38
39#[no_mangle]
40pub extern "C" fn kmain() -> ! {
41	// Mounting first ext2 disk found to DISKNO
42	let disks = disk::discover();
43	for i in disks {
44		if let Ok(ext) = ext2::Ext2::new(i) {
45			kprintln!("Found an ext2 fs");
46			*DISKNO.lock() = Some(ext);
47		}
48	}
49
50	println!("Hello World of {}!", 42);
51	change_color!(Color::Red, Color::White);
52	let workspace_msg = string::String::from(
53		"Press Ctrl-2 to navigate to the second workspace"
54	);
55	println!("{}", workspace_msg);
56	change_color!(Color::White, Color::Black);
57
58	print!("$> ");
59	let mut pid = unsafe { crate::exec_fn!(crate::cli::cli) };
60	loop {
61		// Auto-remove all zombies on pid 0 and relaunch cli if killed
62		let mut status = 0;
63		let ret = sys_waitpid(-1, &mut status, 0);
64		if ret == pid {
65			crate::dprintln!("Term has been killed");
66			print!("$> ");
67			pid = unsafe { crate::exec_fn!(crate::cli::cli) };
68		}
69	}
70}
