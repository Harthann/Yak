87#[derive(Copy, Clone, PartialEq)]
88pub struct Signal {
89	pub sender:  Id,
90	pub sigtype: SignalType,
91	pub wstatus: i32
92}
93
94impl Signal {
95	pub const fn new(pid: Id, sigtype: SignalType, wstatus: i32) -> Self {
96		Self { sender: pid, sigtype: sigtype, wstatus: wstatus }
97	}
98
99	pub fn send_to_pid(
100		pid: Id,
101		sender_pid: Id,
102		sigtype: SignalType,
103		wstatus: i32
104	) -> Result<Id, ErrNo> {
105		let binding = Process::search_from_pid(pid)?;
106		let mut process = binding.lock();
107		Self::send_to_process(&mut *process, sender_pid, sigtype, wstatus);
108		Ok(pid)
109	}
110
111	pub fn send_to_process(
112		process: &mut Process,
113		pid: Id,
114		sigtype: SignalType,
115		wstatus: i32
116	) {
117		let signal = Self::new(pid, sigtype, wstatus);
118		process.signals.push(signal);
119	}
