111#[repr(u8)]
112pub enum SpecialKeyFlag {
113	ShiftLeft  = 0,
114	ShiftRight = 1,
115	Ctrl       = 2,
116	Opt        = 3,
117	CmdLeft    = 4,
118	CmdRight   = 5,
119	Caps       = 6

163fn keyboard_to_ascii(key: u8) -> Option<char> {
164	if is_special(key) || key >= 58 {
165		return None;
166	}
167	// Get actual key pressed
168	let mut charcode: u8 = unsafe { KEYMAP.keys[key as usize] as u8 };
169
170	// If key is alphabetic, check if shift or caps are on
171	if charcode >= b'a' && charcode <= b'z' {
172		if SPECIAL_KEYS.is_enable(SpecialKeyFlag::ShiftLeft as u8)
173			|| SPECIAL_KEYS.is_enable(SpecialKeyFlag::ShiftRight as u8)
174			|| SPECIAL_KEYS.is_enable(SpecialKeyFlag::Caps as u8)
175		{
176			charcode = unsafe { KEYMAP.caps_keys[key as usize] as u8 };
177		}
178	// If key is not alphabetic, switch to cap_keys only if shift is pressed
179	} else if SPECIAL_KEYS.is_enable(SpecialKeyFlag::ShiftLeft as u8)
180		|| SPECIAL_KEYS.is_enable(SpecialKeyFlag::ShiftRight as u8)
181	{
182		{
183			charcode = unsafe { KEYMAP.caps_keys[key as usize] as u8 };
184		}
185	}
186	return Some(charcode as char);
187}

189/// Check if keyboard event is ready
190pub fn keyboard_event() -> bool {
191	io::inb(0x64) & 1 != 0
192}

194use crate::cli::{Input, Termcaps};
195pub fn handle_event() -> Option<(crate::cli::Input, u8)> {
196	let keycode: u8 = io::inb(0x60);
197
198	match keyboard_to_ascii(keycode) {
199		Some(ascii) => {
200			Some((Input::Ascii(ascii), SPECIAL_KEYS.0.load(Ordering::Relaxed)))
201		},
202		None => {
203			let special_keys: &SpecialKeys = unsafe { &KEYMAP.special_keys };
204			match keycode {
205				_ if keycode == special_keys.shift_l[PRESSED] => {
206					SPECIAL_KEYS.enable(SpecialKeyFlag::ShiftLeft as u8);
207					None
208				},
209				_ if keycode == special_keys.shift_r[PRESSED] => {
210					SPECIAL_KEYS.enable(SpecialKeyFlag::ShiftRight as u8);
211					None
212				},
213				_ if keycode == special_keys.ctrl[PRESSED] => {
214					SPECIAL_KEYS.enable(SpecialKeyFlag::Ctrl as u8);
215					None
216				},
217				_ if keycode == special_keys.alt[PRESSED] => {
218					SPECIAL_KEYS.enable(SpecialKeyFlag::Opt as u8);
219					None
220				},
221				_ if keycode == special_keys.special_l[PRESSED] => {
222					SPECIAL_KEYS.enable(SpecialKeyFlag::CmdLeft as u8);
223					None
224				},
225				_ if keycode == special_keys.special_r[PRESSED] => {
226					SPECIAL_KEYS.enable(SpecialKeyFlag::CmdRight as u8);
227					None
228				},
229				_ if keycode == special_keys.caps[PRESSED] => {
230					SPECIAL_KEYS.disable(SpecialKeyFlag::Caps as u8);
231					None
232				},
233
234				_ if keycode == special_keys.shift_l[RELEASED] => {
235					SPECIAL_KEYS.disable(SpecialKeyFlag::ShiftLeft as u8);
236					None
237				},
238				_ if keycode == special_keys.shift_r[RELEASED] => {
239					SPECIAL_KEYS.disable(SpecialKeyFlag::ShiftRight as u8);
240					None
241				},
242				_ if keycode == special_keys.ctrl[RELEASED] => {
243					SPECIAL_KEYS.disable(SpecialKeyFlag::Ctrl as u8);
244					None
245				},
246				_ if keycode == special_keys.alt[RELEASED] => {
247					SPECIAL_KEYS.disable(SpecialKeyFlag::Opt as u8);
248					None
249				},
250				_ if keycode == special_keys.special_l[RELEASED] => {
251					SPECIAL_KEYS.disable(SpecialKeyFlag::CmdLeft as u8);
252					None
253				},
254				_ if keycode == special_keys.special_r[RELEASED] => {
255					SPECIAL_KEYS.disable(SpecialKeyFlag::CmdRight as u8);
256					None
257				},
258				224 => {
259					let keycode: u8 = io::inb(0x60);
260					let special_keys: &SpecialKeys =
261						unsafe { &KEYMAP.special_keys };
262					match keycode {
263						_ if keycode == special_keys.up[PRESSED] => Some((
264							Input::Tcaps(Termcaps::ArrowUP),
265							SPECIAL_KEYS.0.load(Ordering::Relaxed)
266						)),
267						_ if keycode == special_keys.down[PRESSED] => Some((
268							Input::Tcaps(Termcaps::ArrowDOWN),
269							SPECIAL_KEYS.0.load(Ordering::Relaxed)
270						)),
271						_ if keycode == special_keys.left[PRESSED] => Some((
272							Input::Tcaps(Termcaps::ArrowLEFT),
273							SPECIAL_KEYS.0.load(Ordering::Relaxed)
274						)),
275						_ if keycode == special_keys.right[PRESSED] => Some((
276							Input::Tcaps(Termcaps::ArrowRIGHT),
277							SPECIAL_KEYS.0.load(Ordering::Relaxed)
278						)),
279						_ => None
280					}
281				},
282				_ => None
283			}
284		}
285	}
