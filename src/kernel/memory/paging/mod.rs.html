23/// Initialize the paging:
24/// + setup a page_table at the index 768 containing kernel code paddrs and
25/// page_directoy paddr
26/// + reset the initial page_table at index 0 and setup the page_directory
27/// to index every page_tables in memory
28/// + initialize the bitmap of page tables
29/// + refresh tlb to clear the cache of the CPU
30pub fn init_paging() {
31	unsafe {
32		let pd_paddr: PhysAddr =
33			(page_directory.get_vaddr() & 0x3ff000) as PhysAddr;
34		// Claim 1st MiB used by BIOS Real Mode
35		bitmap::physmap_as_mut()
36			.claim_range(0x0, (1024 * 1024) / 0x1000)
37			.expect("Failed to claim BIOS Real Mode 1st MiB");
38		// Claim all memory reserved by grub multiboot
39		multiboot::claim_multiboot();
40		// Claim page_directory
41		bitmap::physmap_as_mut()
42			.claim_range(0x100000, ((pd_paddr / 0x1000) + 1024) as usize)
43			.expect("Failed to claim code pages");
44
45		// Init paging map
46		let kernel_pt_paddr: PhysAddr = bitmap::physmap_as_mut()
47			.get_page()
48			.expect("Failed to get kernel page table");
49		// Use identity mapping to setup kernel page
50		let init_pt_paddr: PhysAddr = pd_paddr + 0x1000;
51		let init_page_tab: &mut PageTable = &mut *(init_pt_paddr as *mut _);
52		init_page_tab.set_entry(
53			KERNEL_BASE >> 22,
54			kernel_pt_paddr | PAGE_WRITABLE | PAGE_PRESENT
55		);
56		refresh_tlb!();
57		// Final mapping
58		let kernel_page_tab: &mut PageTable =
59			&mut *(get_vaddr!(0, KERNEL_BASE >> 22) as *mut _);
60		kernel_page_tab.init();
61		page_directory.set_entry(
62			KERNEL_BASE >> 22,
63			kernel_pt_paddr | PAGE_WRITABLE | PAGE_PRESENT
64		);
65		// Recursive mapping
66		page_directory.set_entry(1023, pd_paddr | PAGE_WRITABLE | PAGE_PRESENT);
67		// Remove init page
68		init_page_tab.clear();
69		page_directory.set_entry(0, 0);
70		refresh_tlb!();
71	}
72}

114macro_rules! get_paddr {
115	($vaddr:expr) => {
116		crate::memory::paging::page_directory
117			.get_page_table(($vaddr as usize) >> 22)
118			.entries[(($vaddr as usize) & 0x3ff000) >> 12]
119			.get_paddr()
120			+ ((($vaddr as usize) & 0xfff) as crate::memory::PhysAddr)
121	};
122}
123
124macro_rules! get_vaddr {
125	($pd_index:expr, $pt_index:expr) => {
126		((($pd_index as usize) << 22) | (($pt_index as usize) << 12))
127			as crate::memory::VirtAddr
128	};
129}

131macro_rules! refresh_tlb {
132	() => {
133		core::arch::asm!("push eax", "mov eax, cr3", "mov cr3, eax", "pop eax")
134	};
135}

137#[allow(unused)]
138macro_rules! enable_paging {
139	($page_directory:expr) => (core::arch::asm!("mov eax, {p}",
140		"mov cr3, eax",
141		"mov eax, cr0",
142		"or eax, 0x80000001",
143		"mov cr0, eax",
144		p = in(reg) (&$page_directory as *const _) as usize););
145}
146
147#[allow(unused)]
148macro_rules! disable_paging {
149	() => {
150		core::arch::asm!("mov ebx, cr0", "and ebx, ~(1 << 31)", "mov cr0, ebx")
151	};
152}
