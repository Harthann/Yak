47pub fn init_memory_addr(
48	addr: VirtAddr,
49	size: usize,
50	flags: u32,
51	kphys: bool
52) -> Result<VirtAddr, ()> {
53	assert!(size % 4096 == 0, "size must be a multiple of 4096");
54	let nb_page: usize = size / 4096;
55
56	if kphys {
57		kalloc_pages_at_addr(addr, nb_page, flags)
58	} else {
59		alloc_pages_at_addr(addr, nb_page, flags)
60	}
61}
62
63pub fn init_memory(
64	size: usize,
65	flags: u32,
66	kphys: bool
67) -> Result<VirtAddr, ()> {
68	assert!(size % 4096 == 0, "size must be a multiple of 4096");
69	let nb_page: usize = size / 4096;
70
71	if kphys {
72		kalloc_pages(nb_page, flags)
73	} else {
74		alloc_pages(nb_page, flags)
75	}
76}

97impl MemoryZone {
98	pub const fn new() -> Self {
99		Self {
100			name:      "",
101			offset:    0,
102			type_zone: TypeZone::Unassigned,
103			size:      0,
104			flags:     0,
105			kphys:     false
106		}
107	}
108
109	pub fn init_addr(
110		offset: VirtAddr,
111		ztype: TypeZone,
112		size: usize,
113		flags: u32,
114		kphys: bool
115	) -> MemoryZone {
116		let mut mz: MemoryZone = MemoryZone {
117			name: "",
118			offset,
119			type_zone: ztype,
120			size,
121			flags,
122			kphys
123		};
124		mz.name = match ztype {
125			TypeZone::Heap => "heap",
126			TypeZone::Stack => "stack",
127			TypeZone::Unassigned => "unassigned",
128			TypeZone::File(name) => name,
129			_ => "Not yet named"
130		};
131		mz.offset = init_memory_addr(offset, size, flags, kphys)
132			.expect("unable to allocate pages for stack");
133		mz
134	}

136	pub fn init(
137		ztype: TypeZone,
138		size: usize,
139		flags: u32,
140		kphys: bool
141	) -> MemoryZone {
142		let mut mz: MemoryZone = MemoryZone {
143			name: "",
144			offset: 0,
145			type_zone: ztype,
146			size,
147			flags,
148			kphys
149		};
150		mz.name = match ztype {
151			TypeZone::Heap => "heap",
152			TypeZone::Stack => "stack",
153			TypeZone::Unassigned => "unassigned",
154			TypeZone::File(name) => name,
155			_ => "Not yet named"
156		};
157		mz.offset = init_memory(size, flags, kphys)
158			.expect("unable to allocate pages for stack");
159		mz
160	}

218use core::fmt;
219impl fmt::Display for MemoryZone {
220	fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
221		let readable: char = if self.flags & READABLE == READABLE {
222			'R'
223		} else {
224			'-'
225		};
226		let writable: char = if self.flags & WRITABLE == WRITABLE {
227			'W'
228		} else {
229			'-'
230		};
231		let executable: char = if self.flags & EXECUTABLE == EXECUTABLE {
232			'X'
233		} else {
234			'-'
235		};
236		let shared: char = if self.flags & MAP_SHARED == MAP_SHARED {
237			'S'
238		} else {
239			'-'
240		};
241		write!(
242			f,
243			"{:#10x} {:#10x} {}{}{}{} [ {} ]",
244			self.offset,
245			self.size,
246			readable,
247			writable,
248			executable,
249			shared,
250			self.name
251		)
252	}
253}

255use core::ops::Drop;
256impl Drop for MemoryZone {
257	fn drop(&mut self) {
258		match self.type_zone {
259			TypeZone::Unassigned => { /* Memory not allocated do nothing */ },
260			_ => {
261				let mut npages = self.size / 4096;
262				// If memory size isn't aligned to page size this means more memory is allocated
263				if self.size % 4096 != 0 {
264					npages += 1
265				}
266				free_pages(self.offset, npages);
267			}
268		}
269	}
270}
