17/// Create a new process from the calling process,
18/// copy stack, heap and registers
19///
20/// Heap contains the prg and the heap allocated
21pub fn sys_fork() -> Pid {
22	unsafe {
23		let running_task: &mut Task = Task::get_running_task();
24		let binding = Process::get_running_process();
25		let mut process: Process = Process::new();
26		let mut new_task: Task = Task::new();
27
28		process.init(&binding);
29
30		let pid = process.pid;
31		{
32			// lock parent in scope
33			let mut parent = binding.lock();
34			process.setup_kernel_stack(parent.kernel_stack.flags);
35			process.setup_stack(
36				parent.stack.size,
37				parent.stack.flags,
38				parent.stack.kphys
39			);
40			process.setup_heap(
41				parent.heap.size,
42				parent.heap.flags,
43				parent.heap.kphys
44			);
45			process.copy_mem(&mut parent);
46		}
47
48		let page_dir: &mut PageDirectory = process.setup_pagination();
49
50		new_task.regs = running_task.regs;
51		new_task.regs.int_no = u32::MAX; // trigger for switch_task
52		new_task.regs.cr3 = get_paddr!(page_dir as *const _);
53		new_task.regs.eax = 0; // New forked process return 0
54
55		new_task.process = KArcm::new(process);
56
57		let mut parent = binding.lock();
58		parent.childs.push(new_task.process.clone());
59		PROCESS_TREE.insert(pid, new_task.process.clone());
60
61		TASKLIST.push_back(new_task);
62		pid
63	}
64}
