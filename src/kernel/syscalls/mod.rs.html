18/// Parameters order: ebx, ecx, edx, esi, edi, ebp
19pub fn syscall_handler(reg: &mut Registers) {
20	if reg.eax > 448 {
21		todo!(); // problem
22	}
23	// TODO: wrapper syscall
24	match reg.eax {
25		_ if reg.eax == Syscall::exit as u32 => sys_exit(reg.ebx as _),
26		_ if reg.eax == Syscall::fork as u32 => reg.eax = sys_fork() as u32,
27		_ if reg.eax == Syscall::waitpid as u32 => {
28			reg.eax =
29				sys_waitpid(reg.ebx as _, reg.ecx as _, reg.edx as _) as u32;
30		},
31		_ if reg.eax == Syscall::getpid as u32 => reg.eax = sys_getpid() as u32,
32		_ if reg.eax == Syscall::getuid as u32 => reg.eax = sys_getuid() as u32,
33		_ if reg.eax == Syscall::kill as u32 => {
34			reg.eax = sys_kill(reg.ebx as _, reg.ecx as _) as u32
35		},
36		_ if reg.eax == Syscall::signal as u32 => {
37			reg.eax = sys_signal(reg.ebx as _, unsafe {
38				core::mem::transmute(reg.ecx as *const ())
39			}) as u32
40		},
41		_ if reg.eax == Syscall::getppid as u32 => {
42			reg.eax = sys_getppid() as u32
43		},
44		_ if reg.eax == Syscall::mmap as u32 => {
45			reg.eax = mmap(reg.ebx as *const mmap::mmap_arg) as u32
46		},
47		_ if reg.eax == Syscall::munmap as u32 => {
48			reg.eax = sys_munmap(reg.ebx as _, reg.ecx as _) as u32
49		},
50		_ if reg.eax == Syscall::wait4 as u32 => {
51			reg.eax = sys_wait4(
52				reg.ebx as _,
53				reg.ecx as _,
54				reg.edx as _,
55				reg.esi as _
56			) as u32
57		},
58		_ => todo!()
59	}
60	// update reg.eax for syscall return
61}
