130impl fmt::Debug for ErrNo {
131	fn fmt(&self, f: &mut fmt::Formatter<'_>) -> fmt::Result {
132		write!(f, "ErrNo: [{:?}] {}", *self as usize, strerror(*self))
133	}
134}

136pub fn strerror(errno: ErrNo) -> &'static str {
137	STRERROR[errno as usize - 1]
